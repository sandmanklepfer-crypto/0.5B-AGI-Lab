## V1. 度量与簇结构（性质 1.2-1）

```python
import torch, torch.nn.functional as F
from transformers import AutoModelForCausalLM, AutoTokenizer

def act(model, tok, text, layer=22):
    ids = tok(text, return_tensors="pt").to("cuda:0")
    with torch.inference_mode():
        h = model(**ids, output_hidden_states=True)
    a = h.hidden_states[layer][0].mean(0).float()  # [896]
    return a / (a.norm() + 1e-12)

def cos(a, b):
    return F.cosine_similarity(a.unsqueeze(0), b.unsqueeze(0)).item()

# 使用
model = AutoModelForCausalLM.from_pretrained("/root/distill_v4c", dtype=torch.bfloat16).to("cuda:0")
tok = AutoTokenizer.from_pretrained("/root/distill_v4c")
print(cos(act(model, tok, "高兴"), act(model, tok, "开心")))   # 期望 ~0.95 (同义)
print(1 - cos(act(model, tok, "高兴"), act(model, tok, "数学"))) # 期望 >0.6 (无关)
# 输出: 同义词距离 d<0.1, 无关词距离 d>0.6 —— 簇结构成立
```

## V2. 有效维（性质 1.3）

```python
import numpy as np, torch

def effective_dim(activations, k=10):
    """输入: [N, 896] 激活矩阵 -> 有效维(谱熵指数)"""
    A = torch.tensor(activations).float().cpu().numpy()
    U, S, Vt = np.linalg.svd(A, full_matrices=False)
    p = S[:k] / S[:k].sum()
    H = -(p * np.log(p + 1e-12)).sum()
    return np.exp(H), S[:k].tolist()

# 健康模型: 概念激活有效维 ~50-100 (分散)
# 死寂模型: 前3奇异值占52% (聚集) —— 有效维下降 = 坍缩信号
```

## V3. 死寂捕获率（性质 4.2-3）

```python
def capture_rate(model, tok, inputs, layer=22, steps=12, thresh=0.05):
    """随机输入被万有吸引子捕获的比例"""
    captured = 0
    for q in inputs:
        ids = tok(q, return_tensors="pt").to("cuda:0")
        gen = ids["input_ids"]
        prev = None
        for _ in range(steps):
            with torch.inference_mode():
                lg = model(gen).logits[:, -1]
            nxt = torch.argmax(lg, dim=-1).unsqueeze(0)
            gen = torch.cat([gen, nxt], dim=1)
            with torch.inference_mode():
                h = model(gen, output_hidden_states=True).hidden_states[layer][0, -1].float()
            h = h / (h.norm() + 1e-12)
            if prev is not None:
                act = 1 - F.cosine_similarity(prev.unsqueeze(0), h.unsqueeze(0)).item()
                if act < thresh:
                    captured += 1; break
            prev = h
    return captured / len(inputs)

# 健康模型: ~0.0 | 死寂模型: ~0.40 (16/40 实测)
```

## V4. Lyapunov 活性（性质 5.2-1）—— 全书最重要函数

```python
def activity(model, tok, q, layer=22, steps=15):
    """活性 λ̄ = 相邻步激活变化率均值。λ̄<0.05 = 死寂预警。"""
    ids = tok(q, return_tensors="pt").to("cuda:0")
    gen = ids["input_ids"]
    prev = None
    lambdas = []
    for _ in range(steps):
        with torch.inference_mode():
            lg = model(gen).logits[:, -1]
        nxt = torch.argmax(lg, dim=-1).unsqueeze(0)
        gen = torch.cat([gen, nxt], dim=1)
        with torch.inference_mode():
            h = model(gen, output_hidden_states=True).hidden_states[layer][0, -1].float()
        h = h / (h.norm() + 1e-12)
        if prev is not None:
            lambdas.append(1 - F.cosine_similarity(prev.unsqueeze(0), h.unsqueeze(0)).item())
        prev = h
    return sum(lambdas) / len(lambdas)

# 实测值: v4c=0.240 (正常) | qwen07b_t=0.046 (死寂) | 7B=0.304 | 32B=0.336-0.361
# 判据: λ̄ < 0.05 → 死寂（生成将开始重复）
```

## V5. 直线锁定检测（性质 5.2-2）

```python
def line_lock_detect(model, tok, q, layer=22, steps=14):
    """死寂轨迹是否为直线锁定（非环）"""
    # 1) 收集轨迹
    ids = tok(q, return_tensors="pt").to("cuda:0")
    gen = ids["input_ids"]
    traj = []
    for _ in range(steps):
        with torch.inference_mode():
            lg = model(gen).logits[:, -1]
        nxt = torch.argmax(lg, dim=-1).unsqueeze(0)
        gen = torch.cat([gen, nxt], dim=1)
        with torch.inference_mode():
            h = model(gen, output_hidden_states=True).hidden_states[layer][0, -1].float()
        traj.append(h / (h.norm() + 1e-12))
    A = torch.stack(traj)
    # 2) PCA 3D 投影
    U, S, V = torch.pca_lowrank(A, q=3)
    P = A @ V  # [steps, 3]
    # 3) 方向一致性: 相邻方向 cos（直线→接近1, 环→接近0）
    dirs = P[1:] - P[:-1]
    dir_cos = F.cosine_similarity(dirs[:-1], dirs[1:], dim=-1).mean().item()
    # 4) 返回判定
    return "直线锁定" if dir_cos > 0.8 else ("旋转环" if dir_cos < 0.3 else "过渡")
# 死寂模型实测: dir_cos=0.8998 → 直线锁定（不是环!）
```

## V6. 纠缠检测（性质 6.2-1）

```python
def entanglement(model_base, model_dist, tok, reason_qs, tpl_qs, layer=22):
    """推理方向与模板方向的夹角 cos —— 纠缠度量"""
    def dir(model, qs):
        d = sum(act(model, tok, q) - act(model_base, tok, q) for q in qs) / len(qs)
        return d / (d.norm() + 1e-12)
    r = dir(model_dist, reason_qs)   # 推理方向
    t = dir(model_dist, tpl_qs)      # 模板方向
    return F.cosine_similarity(r.unsqueeze(0), t.unsqueeze(0)).item()
# 实测: cos=0.8478 → 推理与模板纠缠（线性不可分）
```

```python
def half_fit_judge(ce_history, activity_history):
    """半拟合 vs 全拟合判据"""
    ce_last = ce_history[-1]
    act_last = activity_history[-1]
    if ce_last < 0.01 and act_last < 0.05:
        return "全拟合-死寂（训练过头, 不可逆）"
    if 0.1 < ce_last < 1.3 and act_last > 0.1:
```

## V8. 推理提取（性质 6.2-3）

```python
def extract_reason(base_model, dist_model, alpha=0.3):
    """从蒸馏模型提取推理增量注入基座（推理提升, 模板同步带入）"""
    sd_b, sd_d = base_model.state_dict(), dist_model.state_dict()
    with torch.no_grad():
        for k in sd_b:
            if k in sd_d and "embed" not in k:
                d = sd_d[k].float() - sd_b[k].float()
                sd_b[k] = (sd_b[k].float() + alpha * d).to(torch.bfloat16)
    return base_model
# 实测效果: 三段论从"不会"变"会"(A是C), 但知乎模板同步出现(纠缠)
```

## V9. Hopfield 记忆检索（性质 7）

```python
import numpy as np

class HopfieldMemory:
    """有区分度的记忆吸引子（键=向量, 值=内容, cos检索）"""
    def __init__(self, dim):
        self.dim = dim; self.keys = []; self.vals = []; self.scores = []
    def store(self, key_vec, val, score=0.0):
        k = np.asarray(key_vec, dtype=np.float32).flatten()
        n = np.linalg.norm(k)
        if n < 1e-9: return
        self.keys.append(k / n); self.vals.append(val); self.scores.append(float(score))
    def recall(self, query, topk=1, min_cos=0.5):
        q = np.asarray(query, dtype=np.float32).flatten()
        q = q / (np.linalg.norm(q) + 1e-9)
        K = np.array(self.keys)
        cos = K @ q
        idx = np.argsort(cos)[::-1][:topk]
        return [(float(cos[i]), self.vals[i]) for i in idx if cos[i] >= min_cos]
# 设计原则: 键间隔 > 检索阈值(min_cos), 否则混淆 —— 区分度可调
```
